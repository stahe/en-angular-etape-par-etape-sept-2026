    // =========================================================================
// 10-components/src/app/doctor-map/carte-medecin.ts - a ENFANT component
// -------------------------------------------------------------------------
// It receives its data from its parent via ENTRÉES (inputs), declared
// using input(): each input is a read-only SIGNAL, which follows the
// value passed by the parent. Their type is checked at compile time:
// <app-carte-medecin [medecin]="42"> would be rejected.
//
// Golden rule: a component must not modify its inputs. The data
// flows from the parent to the child; to request a change,
// the child emits an event (see Example 11).
// =========================================================================

import { booleanAttribute, Component, computed, input } from '@angular/core';
import { Medecin } from '../medecin';

@Component({
   // the selector: the tag that represents this component within its parent
  selector: 'app-carte-medecin',
  templateUrl: './carte-medecin.html',
  styleUrl: './carte-medecin.css',
})
export class CarteMedecin {
   // input.required: The parent DOIT provides this entry
  readonly medecin = input.required<Medecin>();
  readonly rang = input.required<number>();
   // input (default value): an optional input. booleanAttribute
   // allows you to simply write <app-carte-medecin compact> for true
  readonly compact = input(false, { transform: booleanAttribute });

   // a computed property based on an input: recalculated if the parent changes it
  protected readonly initiales = computed(() => `${this.medecin().prenom[0]}${this.medecin().nom[0]}`);
}
    • Line 19: the selector: the parent will write [<app-carte-medecin>];
    • lines 25–29: the component’s entries ([inputs])—the data passed to it by its parent. [input()] creates a read-only signal that tracks the value passed by the parent:
      • lines 25–26: [input.required<T>()]: a required input. Omitting [[medecin]] from the parent’s template results in a compilation error;
      • Line 29: [input(valeur par défaut)]: an optional entry. The [transform] option converts the received value: [booleanAttribute] checks for the presence of the attribute ([<app-carte-medecin compact>], with no value) into a [true], as with the attribute HTML [disabled];
    • line 32: a [computed] based on an entry: recalculated if the parent changes the entry.

    The inputs are not [protected]: they are part of the component’s public interface, the one its parent uses.

    Golden rule: a component never modifies its inputs (which are read-only, anyway). Data flows from the parent to the child; to request a change, the child emits an event (Example 11).

    // =========================================================================
// 11-outputs-model/src/app/counter/compteur.ts
// -------------------------------------------------------------------------
// A child that PRÉVIENT its parent: output() declares an event that the
// component can emit, along with the type of its data. The parent listens to it
// as a HTML event: <app-counter (increment)="..." (limit)="...">.
// =========================================================================

import { Component, input, output, signal } from '@angular/core';

@Component({
  selector: 'app-compteur',
  template: `
    <div class="d-flex align-items-center gap-2 mb-2">
      <span class="w-25">{{ libelle() }}</span>
      <button class="btn btn-sm btn-outline-primary" (click)="plus()">+1</button>
      <span class="badge text-bg-secondary">{{ valeur() }} / {{ max() }}</span>
    </div>
  `,
})
export class Compteur {
  readonly libelle = input.required<string>();
  readonly max = input.required<number>();

   // two events: increment (with the new value) and limit (without data)
  readonly incremente = output<number>();
  readonly limite = output<void>();

  protected readonly valeur = signal(0);

  protected plus(): void {
    if (this.valeur() >= this.max()) {
      this.limite.emit();
      return;
    }
    this.valeur.update((v) => v + 1);
    this.incremente.emit(this.valeur());
  }
}
    • lines 13–19: the component’s template, written in the decorator (property [template], enclosed in grave accents);
    • lines 26–27: [output<T>()] declares an output: an event that the component can emit, along with the type of its data. [limite] has no data ([void]);
    • lines 33 and 37: [emit(donnee)] emits the event; the parent receives the data.

    Avoid naming an output after a browser event ([change], [click]...): browser events emitted by the component’s elements propagate up to its host element, and the parent would also receive them.

    // =========================================================================
// 11-outputs-model/src/app/text-field/champ-texte.ts
// -------------------------------------------------------------------------
// A "custom" form field (label + field + tooltip + character counter
// ) that accepts a link in both directions, such as DEUX:
//      <app-text-field [(valeur)]="name" label="Name" [max]="20" />
//
// model() creates a signal MODIFIABLE linked to the parent: LIRE provides the
// the parent’s value; the ÉCRIRE (set) emits the event valeurChange, which the
// syntax [( )] translates into an update to the parent’s signal. The data
// therefore remains the property of the parent.
// =========================================================================

import { Component, input, model } from '@angular/core';

// a counter shared by all instances: a unique identifier per
// field, to link <label for> and <input id>
let prochainNumero = 0;

@Component({
  selector: 'app-champ-texte',
  template: `
    <div class="mb-3">
      <label class="form-label" [for]="id">{{ libelle() }}</label>
       <!-- [value]: the parent’s value; (input): each keystroke replaces it -->
      <input class="form-control" [id]="id" [value]="valeur()" [attr.maxlength]="max()"
             [class.is-invalid]="valeur().length === max()" (input)="saisir($event)">
      <div class="form-text">{{ aide() }} ({{ valeur().length }} / {{ max() }})</div>
    </div>
  `,
})
export class ChampTexte {
  readonly valeur = model.required<string>();
  readonly libelle = input.required<string>();
  readonly max = input.required<number>();
  readonly aide = input('');

  protected readonly id = `champ-texte-${++prochainNumero}`;

  protected saisir(evenement: Event): void {
    this.valeur.set((evenement.target as HTMLInputElement).value);
  }
}
    • Line 33: [model()] creates an editable field linked to the parent. LIRE returns the parent’s value; ÉCRIRE ([set], line 41) changes the value; ET emits the event [valeurChange] (the template name followed by [Change]). The parent that writes [[(valeur)]="signal"] thus has its signal updated;
    • line 18: a module variable (outside the class) is shared by all instances of the component: a counter that assigns a unique identifier to each field (line 38), to link [<label for>] and [<input id>];
    • lines 26–27: [[value]] displays the parent’s value; [(input)] replaces it with each keystroke (line 41). [[attr.maxlength]] is an attribute binding to HTML (the attribute [maxlength]), not a property binding.

// -------------------------------------------------------------------------
// Two ways for a child to send information back up:
//    1. trigger an event (output), which the parent listens for using ( );

  protected readonly message = signal('');
  protected readonly prenom = signal('Marie');
  protected readonly nom = signal('PELISSIER');

    this.prenom.set('');
    this.nom.set('');
  }
}

// 12-projection/src/app/tableau/tableau.ts - an array GÉNÉRIQUE
// -------------------------------------------------------------------------
// Array<T> is a generic class: it displays rows of any
// type T. It knows how to iterate over the rows, but not how to display UNE
// row: the parent tells it how, by passing it a MODÈLE row
// (an <ng-template #line>, a piece of HTML that isn’t displayed as

// contentChild('row'): the <ng-template #row> placed by the parent between

// =========================================================================

import { NgTemplateOutlet } from '@angular/common';

// what the array passes to the model for each row
export interface ContexteLigne<T> {
  $implicit: T; // the value retrieved by let-row
  index: number; // retrieved by let-i="index"
}

     // called once, immediately after the component is first displayed

    this.journal.emit('Horloge : ngOnDestroy - minuterie arrêtée');
  }

    3.5. Example 14: Services and Dependency Injection

    A service is a class that groups together data and code shared by multiple components: server access, the logged-in user, a shopping cart, etc. Components do not create services (never [new PanierService()]): they request them from [Angular], using the [inject()] function. Depending on where it is provided, a service is therefore shared to a greater or lesser extent. This example shows three levels.

    // =========================================================================
// 16-confirmation/src/app/confirmation/confirmation.service.ts
// -------------------------------------------------------------------------
// “Are you sure?”: window.confirm() exists, but you can neither
// style it or translate its buttons. It is replaced by a window
// Bootstrap window (fenetre-confirmation.ts), controlled by this service.
//
// Tip: confirm(message) returns a PROMESSE, which resolves to true (Yes)
// or false (No) when the user responds. The calling code is written
// just as simply as with window.confirm:
//
//      if (await this.confirmation.confirmer('Delete this doctor?')) { ... }
//
// providedIn: 'root': a single service for the entire application, so a
// single window, created once in the root component.
// =========================================================================

import { Injectable, signal } from '@angular/core';

@Injectable({ providedIn: 'root' })
export class ConfirmationService {
  private readonly _message = signal<string | null>(null); // null: window closed
  readonly message = this._message.asReadonly();
  private repondre: ((oui: boolean) => void) | null = null;

  confirmer(texte: string): Promise<boolean> {
    this.repondre?.(false); // Is a question still open? It defaults to "no"
    this._message.set(texte);
    return new Promise<boolean>((resolve) => {
      this.repondre = resolve;
    });
  }

  reponse(oui: boolean): void {
    this._message.set(null);
    const r = this.repondre;
    this.repondre = null;
    r?.(oui);
  }
}
    • line 20: a single service for the entire application; therefore, there will be only one window, placed once and for all in the root component;
    • line 22: the displayed message; [null] means that the window is closed;
    • line 23: [asReadonly()]: components can read the message, but only the service can modify it;
    • line 24: the function that will resolve the current promise, which is awaiting a response;
    • lines 26–32: [confirmer] opens the window (line 28) and returns a promise. Its function [resolve] is set aside (line 30): the promise will not be resolved until later, when the user clicks a button;
    • line 27: if a question were still open, it would be answered “no”: its promise will not remain pending forever;
    • lines 34–39: [reponse] is called by the window: it closes the window (line 35) and then resolves the promise with the user’s response (line 38).

    This is the benefit of the promise: the code that asks the question writes it as a call to [window.confirm()], waiting for the response with [await], without a callback function.

  nom: string;

  selector: 'app-root',

  templateUrl: './app.html',
})
export class App {

  protected readonly contacts = signal<Contact[]>([

    { id: 2, nom: 'Alice DURAND' },
    { id: 3, nom: 'Luc GERMAN' },
  ]);

   // An async method: it “waits” for the user’s response
  protected async supprimer(c: Contact): Promise<void> {

      this.message.set('Suppression abandonnée.');
      return;

    this.contacts.update((liste) => liste.filter((x) => x.id !== c.id));

    if (await this.confirmation.confirmer(`Supprimer les ${this.contacts().length} contacts ?`)) {
      this.contacts.set([]);
      this.message.set('Tous les contacts ont été supprimés.');
    }
  }
}
