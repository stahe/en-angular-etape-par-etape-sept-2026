// 01-first-application/src/app/app.ts - a first component
// -------------------------------------------------------------------------
// An Angular component is a CLASSE TypeScript, decorated with @Component:
//    - selecto   r : the HTML tag that represents it (<app-root>);
//    - templateUrl: the sound file HTML (app.html);
//    - styleUr   l : the sound file CSS (app.css), which applies
//                    to this component.
// The class’s properties and methods can be used in the template.
// =========================================================================

import { Component, signal } from '@angular/core';

@Component({
  selector: 'app-root',
  templateUrl: './app.html',
  styleUrl: './app.css',
})
export class App {
   // "Ordinary" data: the template can display it
  protected readonly titre = 'Bienvenue dans Angular';
  protected readonly langages = ['TypeScript', 'HTML', 'CSS'];

   // signal(...): a RÉACTIVE data item. Its value is read via the APPELANT
   // (counter()), and it is changed using set() or update(). Angular then updates
   // AUTOMATIQUEMENT to the parts of the page that display it.
  protected readonly compteur = signal(0);
  protected readonly heure = signal(new Date().toLocaleTimeString());

   // a method called by the button in the template
  protected incrementer(): void {
    this.compteur.update((n) => n + 1);
    this.heure.set(new Date().toLocaleTimeString());
  }
}

    // =========================================================================
// 02-signals/src/app/app.ts - signal, computed, effect, linkedSignal
// -------------------------------------------------------------------------
// A small price calculator TTC:
//    - signa      l : a reactive value, which is read by calling it and
//                     which is modified using set() / update();
//    - compute    d : a value CALCULÉE derived from other signals,
//                     automatically recalculated when they change;
//    - effec      t : code executed whenever the signals it
//                     reads (a “side effect”);
//    - linkedSignal: a modifiable signal, but REMIS with a value calculated
//                     when another signal changes.
// =========================================================================

import { Component, computed, effect, linkedSignal, signal, untracked } from '@angular/core';
// FormsModule provides [(ngModel)], which links a field to a signal (see example 05)
import { FormsModule } from '@angular/forms';

@Component({
  selector: 'app-root',
  imports: [FormsModule],
  templateUrl: './app.html',
})
export class App {
  protected readonly libelle = signal('Stylo');
  protected readonly prixHT = signal(2.5);
  protected readonly tauxTVA = signal(20); // as a percentage

   // linkedSignal: the quantity resets to 1 every time the item changes,
   // but the user can modify it in the meantime
  protected readonly quantite = linkedSignal({ source: this.libelle, computation: () => 1 });

   // computed: never assigned directly, always recalculated. Angular knows
   // which signals it depends on (the ones it LIT): it recalculates it when one
   // of them changes, and only in that case
  protected readonly totalHT = computed(() => this.quantite() * this.prixHT());
  protected readonly montantTVA = computed(() => (this.totalHT() * this.tauxTVA()) / 100);
  protected readonly totalTTC = computed(() => this.totalHT() + this.montantTVA());

   // a euro amount formatter, in the French style (1,234.50 €)
  private readonly euros = new Intl.NumberFormat('fr-FR', { style: 'currency', currency: 'EUR' });

   // the history of totals TTC, populated by the effect below
  protected readonly historique = signal<string[]>([]);
  private ancienTTC: number | null = null;

  constructor() {
     // effect: executed once, then each time
     // totalTTC (the only signal it reads)
    effect(() => {
      const ttc = this.totalTTC();
      document.title = `TTC : ${this.format(ttc)}`;
      if (this.ancienTTC !== null) {
        const ligne = `${this.format(this.ancienTTC)} -> ${this.format(ttc)}`;
         // untracked: modify history SANS to make it a dependency of the effect
        untracked(() => this.historique.update((h) => [ligne, ...h].slice(0, 5)));
      }
      this.ancienTTC = ttc;
    });
  }

  protected format(montant: number): string {
    return this.euros.format(montant);
  }

  protected reinitialiser(): void {
    this.libelle.set('Stylo');
    this.prixHT.set(2.5);
    this.tauxTVA.set(20);
    this.quantite.set(1);
  }
}
    • line 15: the responsiveness functions, all provided by [@angular/core];
    • Lines 21 and 17: The template uses [[(ngModel)]], which comes from the [FormsModule] module. A component must import (array [imports]) everything its template uses: other components, directives, and pipes. The compiler reports an error if this is omitted;
    • lines 25–27: three signals, [signal(valeur initiale)]. Their type is inferred from the initial value: [signal('Stylo')] is a string signal, [signal(2.5)] is a numeric signal;
    • line 31: a [linkedSignal] is a modifiable signal (it can be written to, just like an ordinary signal), but it is linked to another signal, its source: whenever the source changes (in this case, the item description), its value is reset to the value calculated by [computation] (in this case, 1). When the item changes, the quantity resets to 1;
    • lines 36–38: a [computed] is a value calculated from other signals. It is never written; it is read as a signal ([this.totalHT()]). [Angular] knows which signals it depends on, because it records the ones the function reads during the calculation. It is recalculated when one of them changes, and only in that case: reading [totalTTC()] ten times without anything changing results in only a single calculation. A [computed] may depend on other [computed] values (line 38);
    • line 41: a French-style amount formatter (1,234.50 €). Example 09 will show how to do this with [Angular], the pipes;
    • lines 50–59: a [effect] is a function executed once, then each time the signals it reads change (here, the single [totalTTC]). This is used for side effects: performing actions on something other than the page (the tab title, line 52; the browser’s local storage; a log);
      • line 56: the effect adds an entry to the history. [untracked(...)] performs this update without making [historique] a dependency of the effect; otherwise, the effect would read [historique] (in [update]) and would be re-triggered every time the history changes... which it itself causes;
      • A [effect] is created in the component’s constructor (or when a property is initialized): this is what [Angular] calls the injection context, which we’ll discuss further when covering services (Example 14). It stops on its own when the component is destroyed;
    • Lines 66–71: [set()] replaces the value of a signal. The quantity is a [linkedSignal]: it can be written just like the others.

    Avoid writing into a signal using a [effect] when a [computed] is sufficient: a [computed] describes a value, while a [effect] describes an action. Here, the history depends on the successive values of the total, which a [computed] cannot express.

    2.6. Example 06: Signal-Based Forms

    [Angular] 22 introduces a new way to write forms, entirely based on signals: signal-based forms ([Signal Forms], package [@angular/forms/signals]). The principle:

    • the form template is the UN signal, which contains an object with one property per field;

// -------------------------------------------------------------------------
// The modern way (Angular 22) to write a form:
//    1. MODÈLE: a signal that contains an object, with one property per
//       field (the form reads from and writes to this signal);

  titre: string;
  prenom: string;
  nom: string;
  age: number;
  newsletter: boolean;
  email: string;
  pays: string;
  naissance: string;
  presentation: string;
}

  titre: 'Mme',
  prenom: 'Marie',
  nom: 'PELISSIER',
  age: 30,
  newsletter: false,
  email: '',
  pays: 'fr',
  naissance: '',
  presentation: '',
};

  selector: 'app-root',
  imports: [FormField, JsonPipe],

   // "newsletter" checkbox is selected (valueOf reads the value of a AUTRE field)

// -------------------------------------------------------------------------
// A pipe is a class annotated with @Pipe, whose transform() method
// receives the value placed on the bar (and, optionally,
// parameters) and returns what should be displayed:
//      {{ time slot | time          }} -> "8:00 - 8:20"
//      {{ time slot | time: 'start' }} -> "8:00"
// A pipe is "pure" by default: Angular only calls transform() if the
// value or parameters have CHANGÉ (same object = same result).

import { Pipe, PipeTransform } from '@angular/core';

export interface Creneau {
  hdebut: number;
  mdebut: number;
  hfin: number;
  mfin: number;
}

// "8:05" rather than "8:5"
function heure(h: number, m: number): string {
  return `${h}:${String(m).padStart(2, '0')}`;
}

@Pipe({ name: 'horaire' })
export class HorairePipe implements PipeTransform {
  transform(c: Creneau, partie: 'tout' | 'debut' | 'fin' = 'tout'): string {
    switch (partie) {
      case 'debut':
        return heure(c.hdebut, c.mdebut);
      case 'fin':
        return heure(c.hfin, c.mfin);
      default:
        return `${heure(c.hdebut, c.mdebut)} - ${heure(c.hfin, c.mfin)}`;
    }
  }
}
